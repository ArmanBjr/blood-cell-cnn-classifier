\section{مقدمه}

تشخیص نوع گلبول‌های سفید خون (\lr{White Blood Cells}) یکی از مراحل مهم در آزمایشگاه‌های پزشکی است و به پزشکان کمک می‌کند تا بیماری‌هایی مانند عفونت، آلرژی و سرطان خون را شناسایی کنند. روش سنتی این کار توسط متخصص انسانی و زیر میکروسکوپ انجام می‌شود که وقت‌گیر و خطاپذیر است.

در این پروژه، از شبکه‌های عصبی کانولوشنی (\lr{CNN}) برای دسته‌بندی خودکار تصاویر میکروسکوپی گلبول‌های سفید به چهار نوع اصلی استفاده می‌کنیم. دو رویکرد اصلی بررسی می‌شود: آموزش یک \lr{CNN} از پایه، و استفاده از یادگیری انتقالی با مدل‌های از پیش آموزش‌دیده روی \lr{ImageNet}.

\subsection{ساخت \lr{Data Generator}}

برای هر سه بخش یک \lr{generator} مناسب با \lr{ImageDataGenerator} کرس ساخته می‌شود. \lr{generator} آموزش شامل افزایش داده و \lr{shuffle} است؛ در حالی که \lr{generator}های اعتبارسنجی و آزمون فقط نرمال‌سازی دارند.

\textbf{چرا \lr{train} باید \lr{shuffle} شود ولی \lr{validation} و \lr{test} نه؟}

\begin{itemize}
    \item \lr{train}: اگر \lr{shuffle} نشود، هر \lr{batch} ممکن است فقط از یک کلاس باشد. این باعث می‌شود گرادیان‌های به‌روزرسانی وزن‌ها غیرقابل اطمینان شوند و مدل آموزش ضعیفی ببیند.
    \item \lr{validation} و \lr{test}: \lr{shuffle} نمی‌شوند تا نتایج ارزیابی در هر اجرا یکسان و قابل مقایسه باشند، و بتوان پیش‌بینی‌ها را دقیقاً به تصاویر متناظر نسبت داد.
\end{itemize}

\section{مدل \lr{CNN} از پایه}

\subsection{معماری}

در این فاز یک شبکه‌ی \lr{CNN} ساده از پایه طراحی و آموزش داده شد، بدون هیچ‌گونه وزن از پیش آموزش‌دیده، \lr{Dropout}، \lr{BatchNorm} یا افزایش داده. هدف اصلی داشتن یک خط پایه (\lr{baseline}) برای مقایسه‌ی فازهای بعدی است.

معماری شبکه شامل چهار بلوک کانولوشنی است که هر کدام از یک لایه‌ی \lr{Conv2D} با فعال‌سازی \lr{ReLU} و یک \lr{MaxPooling} تشکیل شده‌اند. در انتها دو لایه‌ی متصل کامل قرار دارد که خروجی نهایی چهار کلاس را با \lr{Softmax} پیش‌بینی می‌کند:

\[
\text{ورودی}\ (128 \times 128 \times 3)
\;\xrightarrow{\text{Conv32+Pool}}\;
\xrightarrow{\text{Conv64+Pool}}\;
\xrightarrow{\text{Conv128+Pool}}\;
\xrightarrow{\text{Conv128+Pool}}\;
\xrightarrow{\text{Flatten}}\;
\xrightarrow{\text{Dense64}}\;
\xrightarrow{\text{Dense4}}
\]

\textbf{شکاف اعتبارسنجی–آزمون:}

\textbf{بهترین و بدترین کلاس‌ها:}
\lr{Lymphocyte} با \lr{F1} برابر \lr{۰.۹۶} و \lr{AUC} برابر \lr{۰.۹۹۹۶} بهترین عملکرد را داشت. این کلاس دارای هسته‌ای بزرگ و گرد و تیره است که از نظر بصری کاملاً متمایز است. در مقابل، \lr{Eosinophil} و \lr{Neutrophil} با \lr{F1} حدود \lr{۰.۶۹} و \lr{۰.۷۱} بدترین عملکرد را داشتند. هر دوی این‌ها گرانولوسیت هستند و هسته‌ای چندلوبه‌ای دارند؛ تنها تفاوت اصلی آن‌ها در نوع و رنگ دانه‌های داخل سلول است که در تصاویر با اندازه کوچک‌شده‌ی \lr{128×128} بخشی از اطلاعات آن‌ها از دست می‌رود.

\textbf{نتیجه‌گیری:}
این مدل پایه با \lr{۷۶۵٬۴۴۴} پارامتر و بدون هیچ تکنیک تنظیم‌سازی، به دقت آزمون \lr{۷۹.۵۳}\% و \lr{AUC} ماکرو \lr{۹۵.۵۲}\% رسید. این اعداد مبنای مقایسه برای فازهای بعدی هستند که در آن‌ها از افزایش داده، \lr{BatchNorm}، \lr{Dropout} و یادگیری انتقالی استفاده خواهد شد.

\section{مدل \lr{CNN} بهبودیافته}

\subsection{تغییرات نسبت به مدل پایه}

در این فاز، معماری مدل پایه‌ی فاز دوم ثابت نگه داشته شد و سه دسته تغییر روی آن اعمال گردید:

\begin{enumerate}
    \item \textbf{افزایش داده (\lr{Data Augmentation}):} پنج تبدیل تصادفی تنها در مرحله‌ی آموزش اعمال شد: چرخش افقی و عمودی (\lr{RandomFlip})، چرخش تصادفی تا \lr{۱۵} درجه (\lr{RandomRotation})، زوم تصادفی تا \lr{۱۰}\% (\lr{RandomZoom})، جابجایی تا \lr{۱۰}\% در هر جهت (\lr{RandomTranslation})، و تغییر روشنایی ±\lr{۲۰}\% (\lr{RandomBrightness}).

    \item \textbf{\lr{Batch Normalization}:} لایه‌های \lr{BN} پس از هر لایه‌ی کانولوشن و پیش از تابع فعال‌سازی درج شدند تا آمار ورودی را تنظیم کرده و همگرایی پایدارتری فراهم شود.

    \item \textbf{\lr{Dropout}:} لایه‌ی \lr{Dropout} با نرخ‌های مختلف پس از لایه‌های متراکم افزوده شد تا از بیش‌برازش جلوگیری شود.
\end{enumerate}

معماری پایه شامل \lr{۳} بلوک کانولوشن با \lr{32}، \lr{64}، و \lr{128} فیلتر، به همراه یک لایه‌ی متراکم \lr{256} نورونی، بدون تغییر باقی ماند.

\subsection{علت کاهش عملکرد نسبت به فاز دوم}

علی‌رغم دقت اعتبارسنجی بالاتر (\lr{۹۹.۵۰}\% در برابر \lr{۹۴.۵۷}\%)، دقت آزمون از \lr{۷۹.۵۳}\% به \lr{۷۲.۳۸}\% کاهش یافت. دو دلیل اصلی این پدیده عبارتند از:

\begin{enumerate}
    \item \textbf{جابجایی توزیع (\lr{Distribution Shift}):} پوشه‌های \lr{TRAIN} و \lr{TEST} دارای ویژگی‌های تصویری متفاوتی هستند (روشنایی، کنتراست، رنگ‌بندی پس‌زمینه). مدل فاز سوم با افزایش داده‌ی قوی‌تر به توزیع آموزشی بهتر تطبیق یافت، اما همین امر تعمیم‌پذیری روی توزیع متفاوت آزمون را کاهش داد. افزایش داده طراحی‌شده برای تنوع درون‌توزیعی است، نه پوشش‌دادن به تفاوت‌های بین‌توزیعی.

    \item \textbf{معیار اعتبارسنجی گمراه‌کننده:} دقت اعتبارسنجی بالا (\lr{99.5}\%) تصویر خوش‌بینانه‌ای داد، چون زیرمجموعه‌ی اعتبارسنجی از همان \lr{TRAIN} جدا شده و با توزیع آزمون فرق دارد. پس از رفع باگ \lr{validation\_split} در فاز قبل، دقت اعتبارسنجی بسیار بالا شد، اما این به‌تنهایی تضمین‌کننده‌ی دقت آزمون نیست.
\end{enumerate}

در مجموع، اگرچه تکنیک‌های افزایش داده، \lr{BN}، و \lr{Dropout} به بهبود آموزش کمک کردند، مشکل اصلی — جابجایی توزیع میان مجموعه‌های \lr{TRAIN} و \lr{TEST} — با این روش‌ها برطرف نشد. انتظار می‌رود فازهای بعدی که از مدل‌های از‌پیش‌آموزش‌دیده‌ی بزرگ‌تر بهره می‌برند، این شکاف را کاهش دهند.

\section{یادگیری انتقالی با \lr{MobileNetV2}}

در فازهای قبل بهترین دقت آزمون به \lr{72.38\%} رسید. یکی از دلایل اصلی این عدد، جابجایی توزیع میان مجموعه‌های \lr{TRAIN} و \lr{TEST} بود — ویژگی‌هایی که مدل از داده‌های آموزشی یاد گرفته با آنچه در آزمون دیده می‌شود کاملاً منطبق نیست. مدل‌های از‌پیش‌آموزش‌دیده روی \lr{ImageNet} این مشکل را کاهش می‌دهند چون از همان ابتدا ویژگی‌های پایه‌ای قوی‌تری دارند که خوب‌تر تعمیم می‌یابند.

برای این فاز از \lr{MobileNetV2} استفاده کردیم. این مدل نسبت به گزینه‌های سنگین‌تر مثل \lr{ResNet50} یا \lr{EfficientNet} پارامتر کمتری دارد و روی \lr{GPU} های معمولی راحت‌تر آموزش می‌بیند، در حالی که دقت قابل قبولی هم ارائه می‌دهد.

\subsection{تغییر لایه‌ی خروجی}

\lr{MobileNetV2} در حالت استاندارد برای طبقه‌بندی \lr{1000} کلاس \lr{ImageNet} طراحی شده. برای تطبیق آن با مسئله‌ی ما (\lr{4} کلاس)، لایه‌ی \lr{top} اصلی حذف شد (\lr{include\_top=False}) و یک سر طبقه‌بند سبک جایگزین آن شد:

\begin{enumerate}
    \item \lr{GlobalAveragePooling2D} — خروجی سه‌بُعدی پشته‌ی کانولوشنی را به یک بردار یک‌بُعدی تبدیل می‌کند. در مقایسه با \lr{Flatten}، پارامتر کمتری دارد و به موقعیت مکانی ویژگی‌ها حساسیت کمتری نشان می‌دهد.
    \item \lr{Dropout} — برای جلوگیری از \lr{overfitting} سر طبقه‌بند.
    \item \lr{Dense(4, softmax)} — لایه‌ی خروجی با \lr{4} نورون برای چهار کلاس سلول خونی.
\end{enumerate}

نکته‌ی مهم پیاده‌سازی این است که در تابع \lr{forward pass}، مدل پایه با آرگومان \lr{training=False} فراخوانی می‌شود. این کار لایه‌های \lr{BatchNormalization} درون \lr{MobileNetV2} را در حالت استنتاج نگه می‌دارد — یعنی از میانگین و واریانس تجمیعی استفاده می‌کنند نه از آمار دسته‌ی جاری. بدون این تنظیم، آموزش سر طبقه‌بند با پارامترهای \lr{BN} منجمدشده ناپایدار می‌شود.

\subsection{پیش‌پردازش ورودی}

\lr{MobileNetV2} انتظار دارد پیکسل‌ها در بازه‌ی $[-1, +1]$ باشند، نه $[0, 1]$. به همین دلیل به جای تقسیم ساده بر \lr{255}، از تابع \lr{mobilenet\_v2.preprocess\_input} استفاده شد که پیکسل‌ها را از $[0, 255]$ به $[-1, +1]$ نگاشت می‌کند. این تفاوت ظاهراً جزئی در عمل تأثیر قابل‌توجهی روی کیفیت آموزش دارد، چون وزن‌های \lr{ImageNet} با همین توزیع ورودی بهینه شده‌اند.

\subsection{پروتکل آموزش دو مرحله‌ای}

\subsubsection{مرحله‌ی اول — آموزش سر طبقه‌بند}

در مرحله‌ی اول تمام لایه‌های \lr{MobileNetV2} منجمد (\lr{freeze}) شدند و فقط لایه‌های سر طبقه‌بند آموزش دیدند. نرخ یادگیری $10^{-3}$ انتخاب شد — چون این لایه‌ها از صفر شروع می‌کنند و می‌توان با قدم‌های بزرگ‌تر آموزش داد. هدف این مرحله این است که سر طبقه‌بند ابتدا یاد بگیرد ویژگی‌های \lr{MobileNetV2} را تفسیر کند، بدون اینکه وزن‌های از‌پیش‌آموزش‌دیده را خراب کند.

\subsubsection{مرحله‌ی دوم — \lr{Fine-tuning} لایه‌های بالایی}

پس از همگرایی مرحله‌ی اول، چند لایه‌ی آخر \lr{backbone} از حالت منجمد خارج (\lr{unfreeze}) شدند و آموزش با نرخ یادگیری بسیار کوچک‌تر ($10^{-5}$) ادامه یافت. دلیل انتخاب نرخ یادگیری کوچک این است که وزن‌های \lr{ImageNet} در این لایه‌ها از قبل خوب هستند؛ فقط نیاز به تطبیق ظریف با دامنه‌ی تصاویر خون داریم. اگر نرخ یادگیری بالا باشد، این وزن‌ها خراب می‌شوند و دقت به سطح تصادفی سقوط می‌کند.

\subsection{تحلیل نهایی}

نتایج این فاز تصویر جالب و در عین حال صادقانه‌ای از محدودیت‌های این دیتاست ارائه می‌دهد.

\textbf{شکاف اعتبارسنجی و آزمون.}
بزرگ‌ترین چیزی که این فاز آشکار کرد همان مشکلی بود که از فاز سوم می‌شناختیم: جابجایی توزیع میان \lr{TRAIN} و \lr{TEST}. \lr{exp1} در اعتبارسنجی به \lr{88.85\%} رسید اما روی آزمون فقط \lr{55.69\%} داد — یعنی مدل با وجود وزن‌های \lr{ImageNet}، به همان دام \lr{overfitting} به توزیع آموزشی افتاد. \lr{exp4} این شکاف را کوچک‌تر کرد: اعتبارسنجی \lr{99.60\%}، آزمون \lr{74.67\%}. اما شکاف هنوز وجود دارد و نشان می‌دهد ریشه‌ی مشکل در خود دیتاست است، نه فقط در معماری مدل.

\textbf{مقایسه با فازهای قبل.}
بهترین مدل این فاز (\lr{exp4}، دقت آزمون \lr{74.67\%}) نسبت به بهترین مدل فاز سوم (\lr{72.38\%}) کمی بهتر شد — حدود \lr{2.3} واحد درصد. این بهبود کوچک‌تر از آن چیزی است که معمولاً از \lr{Transfer Learning} انتظار می‌رود، اما با توجه به جابجایی توزیع دیتاست قابل توضیح است: وزن‌های \lr{ImageNet} کمک کردند ویژگی‌های بهتری استخراج شود، اما مشکل اصلی — که تفاوت توزیع است — با هیچ معماری‌ای به‌تنهایی برطرف نمی‌شود.

\textbf{تحلیل کلاس به کلاس.}
در بهترین مدل (\lr{exp4}) کلاس \lr{LYMPHOCYTE} با \lr{Precision} بالای \lr{0.99} بهترین عملکرد را داشت — شاید چون این سلول ویژگی بصری مشخص‌تری (هسته‌ی گرد بزرگ) دارد که \lr{MobileNetV2} به خوبی آن را یاد گرفت. در مقابل، \lr{MONOCYTE} با \lr{Recall} پایین \lr{0.59} مشکل‌دارترین کلاس بود: مدل اکثر \lr{MONOCYTE} های واقعی را به کلاس‌های دیگر تشخیص داد، که احتمالاً به دلیل شباهت ظاهری این سلول با \lr{LYMPHOCYTE} در نمونه‌های آزمون است.

\textbf{نتیجه‌گیری کلی.}
از میان همه‌ی مدل‌های آزمایش‌شده در این پروژه، \lr{MobileNetV2} با \lr{Fine-tuning} (\lr{exp4}) بهترین دقت آزمون را داشت. اما بهبود نسبت به فاز سوم اندک بود، چون مشکل اصلی این دیتاست — جابجایی توزیع بین مجموعه‌های آموزش و آزمون — یک مشکل ساختاری است که با انتخاب معماری بهتر به‌تنهایی حل نمی‌شود. برای حل اساسی‌تر این مشکل، روش‌هایی مثل \lr{domain adaptation} یا جمع‌آوری داده‌ی آموزشی متنوع‌تر لازم است.
